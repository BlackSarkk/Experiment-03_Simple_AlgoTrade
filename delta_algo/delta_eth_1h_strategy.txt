//@version=5
strategy("Delta Exchange ETH 1H Algo [51 EMA + RSI + Consolidation + Leverage]",
         shorttitle="Delta ETH 1H Algo",
         overlay=true,
         initial_capital=10000,
         default_qty_type=strategy.cash,
         commission_type=strategy.commission.percent,
         commission_value=0.05,
         slippage=1,
         process_orders_on_close=false,
         pyramiding=0)

// =============================================================================
// STRATEGY INPUT PARAMETERS
// =============================================================================
grp_ind   = "1. Technical Indicators"
ema_len   = input.int(51, "EMA Period", minval=5, group=grp_ind, tooltip="Exponential Moving Average period for trend confirmation.")
rsi_len   = input.int(14, "RSI Period", minval=2, group=grp_ind)
rsi_ob    = input.float(65.0, "RSI Overbought Threshold", minval=50.0, maxval=100.0, group=grp_ind)
rsi_os    = input.float(35.0, "RSI Oversold Threshold", minval=0.0, maxval=50.0, group=grp_ind)
atr_len   = input.int(14, "ATR Period", minval=1, group=grp_ind)

grp_cons  = "2. Consolidation & Volume Filters"
cons_len  = input.int(8, "Consolidation Lookback (Candles)", minval=3, group=grp_cons, tooltip="Number of completed candles to verify volatility compression.")
cons_mult = input.float(2.2, "Consolidation ATR Multiplier", minval=0.5, step=0.1, group=grp_cons, tooltip="Maximum allowed price range as a multiple of ATR.")
use_vol   = input.bool(true, "Require Volume Breakout Filter", group=grp_cons)
vol_mult  = input.float(1.0, "Volume Multiplier vs 20 SMA", minval=0.5, step=0.05, group=grp_cons)

grp_risk  = "3. Risk Management & Leverage"
leverage  = input.float(1.0, "Account Leverage Multiplier", minval=1.0, maxval=50.0, step=1.0, group=grp_risk, tooltip="Account leverage (e.g. 1.0 = Spot/1x, 2.0 = 2x, 5.0 = 5x, 10.0 = 10x).")
risk_pct  = input.float(1.5, "Account Risk % Per Trade", minval=0.1, maxval=10.0, step=0.1, group=grp_risk, tooltip="Percentage of current equity risked per trade.")
max_alloc = input.float(50.0, "Max Position Allocation %", minval=5.0, maxval=100.0, step=5.0, group=grp_risk, tooltip="Hard cap on nominal position value relative to equity * leverage.")
rr_ratio  = input.float(1.5, "Risk-to-Reward Target (1 : X)", minval=0.5, step=0.1, group=grp_risk)
swing_len = input.int(8, "Swing High/Low Lookback", minval=2, group=grp_risk, tooltip="Lookback window for structural Swing High / Swing Low stop-loss.")

grp_ui    = "4. Visual Display & Dashboard"
show_cons = input.bool(true, "Highlight Consolidation Zones", group=grp_ui)
show_sl_tp= input.bool(true, "Plot Active SL & TP Lines", group=grp_ui)
show_hud  = input.bool(true, "Show On-Chart Performance HUD Table", group=grp_ui)

// =============================================================================
// INDICATOR CALCULATIONS
// =============================================================================
// 51 EMA
ema51 = ta.ema(close, ema_len)

// 14 RSI
rsi14 = ta.rsi(close, rsi_len)

// 14 ATR
atr14 = ta.atr(atr_len)

// 20-period Volume Moving Average
vol_sma = ta.sma(volume, 20)

// 8-Candle Consolidation Detection
cons_high  = ta.highest(high, cons_len)
cons_low   = ta.lowest(low, cons_len)
cons_range = cons_high - cons_low
is_consolidating = cons_range <= (atr14 * cons_mult)
prior_consolidation = math.max(is_consolidating[1] ? 1 : 0, is_consolidating[2] ? 1 : 0, is_consolidating[3] ? 1 : 0) == 1

// Confirmed Swing Levels (Lookback shifted by 1 to prevent lookahead bias)
swing_low  = ta.lowest(low[1], swing_len)
swing_high = ta.highest(high[1], swing_len)

// RSI Momentum Recovery Context (6-candle lookback)
rsi_was_oversold   = ta.lowest(rsi14, 6) <= rsi_os
rsi_was_overbought = ta.highest(rsi14, 6) >= rsi_ob

// Volume Confirmation
vol_confirmed = not use_vol or (volume >= vol_sma * vol_mult)

// =============================================================================
// ENTRY SIGNAL LOGIC (Completed-Candle Confirmation)
// =============================================================================
// 1. Completed-Candle EMA Crossover
ema_cross_up   = (close[1] <= ema51[1] and close > ema51) or (close > ema51 and close[1] > ema51[1] and open < ema51)
ema_cross_down = (close[1] >= ema51[1] and close < ema51) or (close < ema51 and close[1] < ema51[1] and open > ema51)

// 2. RSI Momentum Filter
rsi_long_valid  = (rsi14 < rsi_ob) and (rsi14 >= 40.0 or rsi_was_oversold)
rsi_short_valid = (rsi14 > rsi_os) and (rsi14 <= 60.0 or rsi_was_overbought)

// 3. Consolidation Breakout Confirmation
cons_valid = prior_consolidation or is_consolidating

// Combined Entry Triggers
long_signal  = ema_cross_up and rsi_long_valid and cons_valid and vol_confirmed
short_signal = ema_cross_down and rsi_short_valid and cons_valid and vol_confirmed

// =============================================================================
// POSITION SIZING & STRATEGY EXECUTION WITH LEVERAGE
// =============================================================================
var float entry_sl = na
var float entry_tp = na

// Long Entry
if (long_signal and strategy.position_size == 0)
    calc_sl = math.min(swing_low, low)
    if (close - calc_sl) < (0.4 * atr14)
        calc_sl := close - (0.85 * atr14)
    
    risk_dist = close - calc_sl
    if (risk_dist > 0)
        entry_sl := calc_sl
        entry_tp := close + (rr_ratio * risk_dist)
        
        // Dynamic Risk Sizing
        risk_budget = strategy.equity * (risk_pct / 100.0)
        raw_qty     = risk_budget / risk_dist
        
        // Leverage & Max Nominal Capacity Cap
        max_alloc_qty = (strategy.equity * leverage * (max_alloc / 100.0)) / close
        final_qty     = math.min(raw_qty, max_alloc_qty)
        
        strategy.entry("Long", strategy.long, qty=final_qty)
        strategy.exit("Exit Long", from_entry="Long", stop=entry_sl, limit=entry_tp)

// Short Entry
if (short_signal and strategy.position_size == 0)
    calc_sl = math.max(swing_high, high)
    if (calc_sl - close) < (0.4 * atr14)
        calc_sl := close + (0.85 * atr14)
        
    risk_dist = calc_sl - close
    if (risk_dist > 0)
        entry_sl := calc_sl
        entry_tp := close - (rr_ratio * risk_dist)
        
        risk_budget = strategy.equity * (risk_pct / 100.0)
        raw_qty     = risk_budget / risk_dist
        
        max_alloc_qty = (strategy.equity * leverage * (max_alloc / 100.0)) / close
        final_qty     = math.min(raw_qty, max_alloc_qty)
        
        strategy.entry("Short", strategy.short, qty=final_qty)
        strategy.exit("Exit Short", from_entry="Short", stop=entry_sl, limit=entry_tp)

// Reset levels when flat
if (strategy.position_size == 0)
    entry_sl := na
    entry_tp := na

// =============================================================================
// CHART PLOTTING & VISUALIZATION
// =============================================================================
// Plot 51 EMA
ema_color = close >= ema51 ? color.new(#10b981, 0) : color.new(#ef4444, 0)
plot(ema51, "51 EMA", color=ema_color, linewidth=2)

// Highlight 8-Candle Consolidation Zones
bgcolor(show_cons and is_consolidating ? color.new(#3b82f6, 92) : na, title="8-Candle Consolidation Zone")

// Plot Active SL / TP Targets
plot(show_sl_tp and strategy.position_size != 0 ? entry_sl : na, "Stop Loss", color=color.new(#ef4444, 0), style=plot.style_linebr, linewidth=2)
plot(show_sl_tp and strategy.position_size != 0 ? entry_tp : na, "Take Profit", color=color.new(#10b981, 0), style=plot.style_linebr, linewidth=2)

// Plot Entry Signal Markers
plotshape(long_signal and strategy.position_size == 0, title="Long Breakout Signal", location=location.belowbar, color=color.new(#10b981, 0), style=shape.triangleup, size=size.small, text="LONG")
plotshape(short_signal and strategy.position_size == 0, title="Short Breakdown Signal", location=location.abovebar, color=color.new(#ef4444, 0), style=shape.triangledown, size=size.small, text="SHORT")

// =============================================================================
// ON-CHART PERFORMANCE HUD TABLE
// =============================================================================
var table hud = table.new(position.top_right, 2, 8, bgcolor=color.new(#0f172a, 10), border_color=color.new(#334155, 0), border_width=1)

if (show_hud and barstate.islast)
    win_rate  = strategy.closedtrades > 0 ? (strategy.wintrades / strategy.closedtrades) * 100.0 : 0.0
    net_pnl   = strategy.netprofit
    pnl_color = net_pnl >= 0 ? #10b981 : #ef4444
    pos_str   = strategy.position_size > 0 ? "LONG (" + str.tostring(strategy.position_size, "#.##") + " ETH)" : strategy.position_size < 0 ? "SHORT (" + str.tostring(math.abs(strategy.position_size), "#.##") + " ETH)" : "FLAT"

    table.cell(hud, 0, 0, "Strategy", text_color=#94a3b8, text_size=size.small, text_halign=text.align_left)
    table.cell(hud, 1, 0, "Delta 1H Algo", text_color=#38bdf8, text_size=size.small, text_halign=text.align_right)

    table.cell(hud, 0, 1, "Leverage", text_color=#94a3b8, text_size=size.small, text_halign=text.align_left)
    table.cell(hud, 1, 1, str.tostring(leverage, "#.0") + "x", text_color=#f8fafc, text_size=size.small, text_halign=text.align_right)

    table.cell(hud, 0, 2, "Net Profit ($)", text_color=#94a3b8, text_size=size.small, text_halign=text.align_left)
    table.cell(hud, 1, 2, "$" + str.tostring(net_pnl, "#,###.00"), text_color=pnl_color, text_size=size.small, text_halign=text.align_right)

    table.cell(hud, 0, 3, "Win Rate (%)", text_color=#94a3b8, text_size=size.small, text_halign=text.align_left)
    table.cell(hud, 1, 3, str.tostring(win_rate, "#.1") + "%", text_color=#f8fafc, text_size=size.small, text_halign=text.align_right)

    table.cell(hud, 0, 4, "Total Trades", text_color=#94a3b8, text_size=size.small, text_halign=text.align_left)
    table.cell(hud, 1, 4, str.tostring(strategy.closedtrades), text_color=#f8fafc, text_size=size.small, text_halign=text.align_right)

    table.cell(hud, 0, 5, "Profit Factor", text_color=#94a3b8, text_size=size.small, text_halign=text.align_left)
    table.cell(hud, 1, 5, str.tostring(strategy.grossprofit / math.max(strategy.grossloss, 1e-6), "#.##"), text_color=#f8fafc, text_size=size.small, text_halign=text.align_right)

    table.cell(hud, 0, 6, "Max Drawdown", text_color=#94a3b8, text_size=size.small, text_halign=text.align_left)
    table.cell(hud, 1, 6, "$" + str.tostring(strategy.max_drawdown, "#,###.00"), text_color=#f59e0b, text_size=size.small, text_halign=text.align_right)

    table.cell(hud, 0, 7, "Active State", text_color=#94a3b8, text_size=size.small, text_halign=text.align_left)
    table.cell(hud, 1, 7, pos_str, text_color=#38bdf8, text_size=size.small, text_halign=text.align_right)
